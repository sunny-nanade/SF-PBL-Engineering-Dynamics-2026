import numpy as np
import matplotlib.pyplot as plt
from scipy.integrate import solve_ivp


g = 9.81
M = 0.5
m = 0.1
l = 0.3
r = 0.05
I = 0.006
J = 0.0005

# PID gains
Kp, Ki, Kd = 150, 2, 30


def dynamics(t, y):
    theta, theta_dot, x, x_dot, integral = y

    error = theta
    d_error = theta_dot
    integral_dot = error

    u = -(Kp * error + Ki * integral + Kd * d_error)

    sin_t = np.sin(theta)
    cos_t = np.cos(theta)

    # Equations (based on your FBD)
    theta_ddot = (M*g*l*sin_t + u*cos_t) / (I + M*l**2)
    x_ddot = (u/r - M*l*theta_ddot*cos_t + M*l*theta_dot**2*sin_t) / (M + 2*m)

    return [theta_dot, theta_ddot, x_dot, x_ddot, integral_dot]



y0 = [0.15, 0, 0, 0, 0]


t_span = (0, 10)
t_eval = np.linspace(0, 10, 1000)

sol = solve_ivp(dynamics, t_span, y0, t_eval=t_eval)

theta = sol.y[0]
theta_dot = sol.y[1]
x = sol.y[2]
x_dot = sol.y[3]
t = sol.t


control = -(Kp * theta + Ki * np.cumsum(theta)*(t[1]-t[0]) + Kd * theta_dot)


T = 0.5*(M+2*m)*x_dot**2 + 0.5*I*theta_dot**2 + 0.5*M*(l*theta_dot)**2
V = M*g*l*(1 - np.cos(theta))
E = T + V


plt.figure()
plt.plot(t, theta, label="Theta (rad)")
plt.plot(t, theta_dot, label="Theta_dot (rad/s)")
plt.plot(t, x, label="Position (m)")
plt.plot(t, x_dot, label="Velocity (m/s)")
plt.title("System States vs Time")
plt.xlabel("Time (s)")
plt.legend()
plt.grid()


plt.figure()
plt.plot(t, T, label="Kinetic Energy")
plt.plot(t, V, label="Potential Energy")
plt.plot(t, E, label="Total Energy")
plt.title("Energy vs Time")
plt.xlabel("Time (s)")
plt.legend()
plt.grid()


plt.figure()
plt.plot(theta, theta_dot)
plt.xlabel("Theta (rad)")
plt.ylabel("Theta_dot (rad/s)")
plt.title("Phase Plot (Theta vs Angular Velocity)")
plt.grid()


plt.figure()
plt.plot(x, x_dot)
plt.xlabel("Position (m)")
plt.ylabel("Velocity (m/s)")
plt.title("Phase Plot (Position vs Velocity)")
plt.grid()

plt.figure()
plt.plot(t, control)
plt.xlabel("Time (s)")
plt.ylabel("Control Input")
plt.title("Control Effort vs Time")
plt.grid()


plt.figure()
plt.plot(t, theta)
plt.xlabel("Time (s)")
plt.ylabel("Tilt Error (rad)")
plt.title("Error vs Time")
plt.grid()


plt.figure()
plt.plot(t, E - E[0])
plt.xlabel("Time (s)")
plt.ylabel("ΔEnergy (J)")
plt.title("Energy Deviation")
plt.grid()


plt.figure()
plt.plot(t, theta, label="Theta")
plt.plot(t, x, label="Position")
plt.plot(t, E, label="Total Energy")
plt.xlabel("Time (s)")
plt.title("Overall System Response")
plt.legend()
plt.grid()

plt.show()


print("Initial Energy:", E[0])
